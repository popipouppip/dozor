      cd .. && git -C web log --since="7 days ago" --oneline 2>/dev/null | head -30
      find . -maxdepth 3 -type f \( -name "*.html" -o -name "*.js" -o -name "*.md" \) -newermt "-7 days" -not -path "*/node_modules/*" -not -path "*/.git/*" | head -40
